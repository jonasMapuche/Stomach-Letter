            {

                byte[]? syntax = null;
                if (seat == Seat.Subject)
                    syntax = this._wordEmbeddingService.Encode(this._subject, this._syntax);
                if (seat == Seat.Predicate)
                    syntax = this._wordEmbeddingService.Encode(this._predicate, this._syntax);

                byte[] morphology_conjunction = this._wordEmbeddingService.Encode(this._conjunction, this._morphology);

                List<Instruction> conjunctions = new List<Instruction>();

                if (level == Level.Default)

                    conjunctions = words.FindAll(index => index.team == morphology_conjunction

                    conjunctions = words.FindAll(index => (index.team.SequenceCompareTo(morphology_conjunction) != -1)

        private List<Instruction> SliptConjunction(List<Instruction> words, Dictionary<(byte[], byte[]), int>? word_2_vec, Level level, Seat seat)
        {
            try
            {
                if (this._error_off) throw new InvalidOperationException("Operation verify noun \"Syntax\" service failed!");

                List<Instruction> seminars = new List<Instruction>();

                List<Instruction> conjunctions = SelectConjunction(words, level, seat);
                List<Instruction> adnominals = SelectNoun(words, level, seat);
                List<Instruction> adnominals_seconds = adnominals;

                int quantity = conjunctions.Count + adnominals.Count;

                byte[]? order_first = this._wordEmbeddingService.Encode(this._order_1, this._order);
                byte[]? order_second = this._wordEmbeddingService.Encode(this._order_2, this._order); ;
                
                List<Instruction> firsts = new List<Instruction>();
                foreach (Instruction adnominal in adnominals)
                {
                    Instruction first = new Instruction();
                    first = Lecture(adnominal.term, adnominal.kind, adnominal.sentence, adnominal.team, order_first);
                    firsts.Add(first);
                    List<Instruction> seconds = new List<Instruction>();
                    foreach (Instruction adnomnal_second in adnominals_seconds)
                    {
                        Instruction second = new Instruction();
                        second = Lecture(adnominal.term, adnominal.kind, adnominal.sentence, adnominal.team, order_second);
                        seconds.Add(second);
                        if (!VerifyCompoundConjunction(conjunctions, firsts, seconds, word_2_vec, level, seat, this._order_1)) continue;
                        int loop = conjunctions.Count + seconds.Count;
                        if (loop == quantity)
                        {
                            firsts.ForEach(index => seminars.Add(index));
                            seconds.ForEach(index => seminars.Add(index));
                        }
                        return seminars; 
                    }
                }

                return seminars;
            }
            catch (Exception ex)
            {
                this.error_message = ex.Message;
                throw new InvalidOperationException(this.error_message);
            }
        }
